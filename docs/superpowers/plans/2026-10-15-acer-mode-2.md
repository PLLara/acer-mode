## Global Constraints

- Plataforma: Windows 11, PHN16-72; `net8.0-windows`; Node/TS não se aplicam.
- Textos da interface e mensagens em PT-BR; identificadores em inglês; comentários em PT-BR (padrão do `AcerMode.cs`).
- Códigos de modo: Quiet `0`, Balanced `1`, Performance `4`, Turbo `5`, Eco `6`.
- Timeout de chamada WMI: 4 s; pausa após falha: 30 s.
- Debounce do botão físico: 600 ms. Debounce da prévia de iluminação: ~120 ms.
- `FanGuard`: limite **> 85 °C** em CPU ou GPU; % personalizado mínimo **30 %**.
- Sensores: 1 s com UI visível; 3 s escondido com ventoinha em Personalizado; parado nos demais casos; histórico de 120 amostras.
- Dados em `%APPDATA%\AcerMode\` (`settings.json`, `events.log`, `fans-dirty`).
- Manifesto `asInvoker` + relançamento elevado; `--demo` e `--shot` nunca pedem UAC.
- Animações: 150–250 ms ease-out; troca de accent 300 ms; desligadas se `SystemParameters.ClientAreaAnimation == false`.
- Nada sai da máquina (sem rede).

```csharp
namespace AcerMode.Core;

// Códigos que a BIOS do PHN16-72 usa em GetGamingMiscSetting(0x0B).
public enum ThermalMode : byte { Quiet = 0, Balanced = 1, Performance = 4, Turbo = 5, Eco = 6 }

public static class ThermalModes
{
    public static readonly ThermalMode[] All =
        { ThermalMode.Quiet, ThermalMode.Balanced, ThermalMode.Performance, ThermalMode.Turbo, ThermalMode.Eco };

    public static bool IsKnown(int code) => code is >= 0 and <= 255 && Array.IndexOf(All, (ThermalMode)code) >= 0;

    public static string Description(ThermalMode m) => m switch
    {
        ThermalMode.Quiet => "Silencioso, ventoinhas baixas",
        ThermalMode.Balanced => "Equilíbrio entre ruído e desempenho",
        ThermalMode.Performance => "Mais potência para jogos e trabalho pesado",
        ThermalMode.Turbo => "Potência máxima, ventoinhas no talo",
        ThermalMode.Eco => "Economia de bateria",
        _ => "",
    };

    public static bool TryParse(string? s, out ThermalMode mode)
    {
        foreach (var m in All)
            if (string.Equals(m.ToString(), s?.Trim(), StringComparison.OrdinalIgnoreCase)) { mode = m; return true; }
        mode = ThermalMode.Balanced;
        return false;
    }
}
```

```csharp
using System.Globalization;
using System.Text.Json;
using System.Text.Json.Serialization;

namespace AcerMode.Core;

public readonly record struct Rgb(byte R, byte G, byte B)
{
    public static Rgb FromHex(string hex) =>
        TryFromHex(hex, out var c) ? c : throw new FormatException("Cor inválida: " + hex);

    public static bool TryFromHex(string? hex, out Rgb color)
    {
        color = default;
        var s = hex?.Trim().TrimStart('#');
        if (s is null || s.Length != 6 || !int.TryParse(s, NumberStyles.HexNumber, CultureInfo.InvariantCulture, out var v))
            return false;
        color = new Rgb((byte)(v >> 16), (byte)(v >> 8), (byte)v);
        return true;
    }

    public string ToHex() => $"#{R:X2}{G:X2}{B:X2}";
}

// Cores ficam legíveis no settings.json ("#FF2D0A").
public sealed class RgbJsonConverter : JsonConverter<Rgb>
{
    public override Rgb Read(ref Utf8JsonReader reader, Type typeToConvert, JsonSerializerOptions options) =>
        Rgb.TryFromHex(reader.GetString(), out var c) ? c : throw new JsonException("Cor inválida");

    public override void Write(Utf8JsonWriter writer, Rgb value, JsonSerializerOptions options) =>
        writer.WriteStringValue(value.ToHex());
}
```

```csharp
namespace AcerMode.Core;

public enum KbEffect : byte { Static = 0, Breathing = 1, Neon = 2, Wave = 3, Shifting = 4, Zoom = 5, Meteor = 6, Twinkling = 7 }
public enum KbDirection : byte { RightToLeft = 1, LeftToRight = 2 }

public static class KbEffects
{
    public static readonly KbEffect[] All = Enum.GetValues<KbEffect>();
    public static bool UsesZones(KbEffect e) => e == KbEffect.Static;
    public static bool UsesColor(KbEffect e) => e is KbEffect.Breathing or KbEffect.Shifting or KbEffect.Zoom or KbEffect.Meteor or KbEffect.Twinkling;
    public static bool UsesSpeed(KbEffect e) => e != KbEffect.Static;
    public static bool UsesDirection(KbEffect e) => e is KbEffect.Wave or KbEffect.Shifting;

    public static string Label(KbEffect e) => e switch
    {
        KbEffect.Static => "Estático", KbEffect.Breathing => "Respirar", KbEffect.Neon => "Neon",
        KbEffect.Wave => "Onda", KbEffect.Shifting => "Alternar", KbEffect.Zoom => "Zoom",
        KbEffect.Meteor => "Meteoro", KbEffect.Twinkling => "Cintilar", _ => e.ToString(),
    };
}

public sealed record LightingProfile
{
    public const int ZoneCount = 4, MaxSpeed = 9, MaxBrightness = 100;

    public KbEffect Effect { get; init; } = KbEffect.Static;
    public Rgb[] Zones { get; init; } = { new(255, 255, 255), new(255, 255, 255), new(255, 255, 255), new(255, 255, 255) };
    public Rgb Color { get; init; } = new(255, 255, 255);
    public int Brightness { get; init; } = MaxBrightness;
    public int Speed { get; init; } = 5;
    public KbDirection Direction { get; init; } = KbDirection.RightToLeft;

    public static LightingProfile Solid(Rgb c, int brightness) =>
        new() { Zones = new[] { c, c, c, c }, Color = c, Brightness = brightness };

    // Garante 4 zonas e valores dentro do que a BIOS aceita.
    public LightingProfile Normalized()
    {
        var zones = new Rgb[ZoneCount];
        for (int i = 0; i < ZoneCount; i++) zones[i] = Zones is not null && i < Zones.Length ? Zones[i] : Color;
        return this with
        {
            Zones = zones,
            Brightness = Math.Clamp(Brightness, 0, MaxBrightness),
            Speed = Math.Clamp(Speed, 0, MaxSpeed),
            Direction = Direction == KbDirection.LeftToRight ? KbDirection.LeftToRight : KbDirection.RightToLeft,
            Effect = Enum.IsDefined(Effect) ? Effect : KbEffect.Static,
        };
    }
}
```

```csharp
namespace AcerMode.Core;

public enum Sensor : byte { CpuTemp = 0x01, CpuFanRpm = 0x02, SystemTemp = 0x03, GpuFanRpm = 0x06, GpuTemp = 0x0A }

public sealed record SensorReading(int? CpuTempC, int? GpuTempC, int? SystemTempC, int? CpuFanRpm, int? GpuFanRpm, DateTime At)
{
    public bool HasTemps => CpuTempC is not null && GpuTempC is not null;

    // Leitura fora da faixa física é tratada como indisponível, nunca como dado.
    public static int? Plausible(Sensor s, int? raw) => raw switch
    {
        null => null,
        int v when s is Sensor.CpuTemp or Sensor.GpuTemp or Sensor.SystemTemp => v is >= 1 and <= 120 ? v : null,
        int v => v is >= 0 and <= 9000 ? v : null,
    };
}

public sealed class SensorHistory
{
    readonly SensorReading[] items;
    int start, count;
    readonly object gate = new();

    public SensorHistory(int capacity = 120) => items = new SensorReading[capacity];

    public void Add(SensorReading r)
    {
        lock (gate)
        {
            items[(start + count) % items.Length] = r;
            if (count < items.Length) count++; else start = (start + 1) % items.Length;
        }
    }

    public SensorReading? Latest { get { lock (gate) return count == 0 ? null : items[(start + count - 1) % items.Length]; } }

    public IReadOnlyList<SensorReading> Snapshot()
    {
        lock (gate)
        {
            var list = new SensorReading[count];
            for (int i = 0; i < count; i++) list[i] = items[(start + i) % items.Length];
            return list;
        }
    }
}
```

`Hardware/Encoders.cs`:

```csharp
namespace AcerMode.Core;

// Tradução pura entre o domínio e os inteiros/bytes da interface WMI AcerGamingFunction.
// Formatos confirmados no PHN16-72: ver spec §3.1.
public static class Encoders
{
    public const uint ThermalSetting = 0x0B;
    public const uint SensorCommand = 0x01;   // 0x02 (outros modelos) devolve zero aqui
    public const uint BothFans = 0x09;        // bitmask CPU (0x01) | GPU (0x08)

    public static bool StatusOk(ulong output) => (output & 0xFF) == 0;

    public static ulong ThermalSet(ThermalMode m) => ((ulong)m << 8) | ThermalSetting;
    public static int ThermalParse(ulong output) => (int)((output >> 8) & 0xFF);

    public static ulong ZoneColor(int zone, Rgb c)
    {
        if (zone is < 0 or >= LightingProfile.ZoneCount) throw new ArgumentOutOfRangeException(nameof(zone));
        return (1UL << zone) | ((ulong)c.R << 8) | ((ulong)c.G << 16) | ((ulong)c.B << 24);
    }

    public static Rgb ZoneColorParse(ulong output) => new((byte)(output >> 8), (byte)(output >> 16), (byte)(output >> 24));

    // [efeito, velocidade, brilho, 0, direção, R, G, B, motor (3 = quatro zonas), 1, 0...]
    public static byte[] Backlight(LightingProfile profile)
    {
        var p = profile.Normalized();
        bool isStatic = p.Effect == KbEffect.Static;
        var c = isStatic ? p.Zones[0] : p.Color;
        return new byte[]
        {
            (byte)p.Effect, (byte)(isStatic ? 0 : p.Speed), (byte)p.Brightness, 0,
            (byte)(isStatic ? 0 : (byte)p.Direction), c.R, c.G, c.B, 3, 1, 0, 0, 0, 0, 0, 0,
        };
    }

    public static uint SensorQuery(Sensor s) => SensorCommand | ((uint)s << 8);
```

```csharp
namespace AcerMode.Core;

// Executa a ação com o valor mais recente: enquanto uma execução está em curso, só o
// último valor postado fica esperando. Evita encher a BIOS de escritas (roda de cor, cliques).
public sealed class LatestWins<T>
{
    readonly Func<T, Task> run;
    readonly object gate = new();
    bool running, hasPending;
    T pending = default!;

    public LatestWins(Func<T, Task> run) => this.run = run;

    public event Action<Exception>? Error;

    public async Task PostAsync(T value)
    {
        lock (gate)
        {
            if (running) { pending = value; hasPending = true; return; }
            running = true;
        }
        var current = value;
        while (true)
        {
            try { await run(current); }
            catch (Exception ex) { Error?.Invoke(ex); }
            lock (gate)
            {
                if (!hasPending) { running = false; return; }
                current = pending; hasPending = false;
            }
        }
    }
}
```

```csharp
        {
```

```csharp
            ThermalMode.Performance => LightingProfile.Solid(new Rgb(150, 70, 255), 100),
            ThermalMode.Quiet => LightingProfile.Solid(new Rgb(0, 190, 255), 45),
            ThermalMode.Eco => LightingProfile.Solid(new Rgb(40, 220, 90), 70),
            _ => LightingProfile.Solid(new Rgb(40, 120, 255), 100),
        },
        Fans = new FanSettings(),
    };

    {
        if (!Profiles.TryGetValue(m, out var p)) Profiles[m] = p = DefaultProfile(m);
        return p;
    }

    public LightingProfile LightingFor(ThermalMode m) =>
        IndependentLighting && GlobalLighting is not null ? GlobalLighting : Profile(m).Lighting;

    public void SetIndependentLighting(bool on, ThermalMode active)
    {
        if (on && !IndependentLighting) GlobalLighting = Profile(active).Lighting;
        IndependentLighting = on;
    }

    public void Normalize()
    {
        Profiles ??= new();
        foreach (var m in ThermalModes.All)
        {
            var p = Profile(m);
            p.Lighting = (p.Lighting ?? DefaultProfile(m).Lighting).Normalized();
            p.Fans = (p.Fans ?? new FanSettings()).Normalized();
        }
        foreach (var key in Profiles.Keys.Where(k => !ThermalModes.IsKnown((int)k)).ToList()) Profiles.Remove(key);
```

```csharp
namespace AcerMode.Core;

public sealed class SettingsStore

    public static string DefaultDirectory =>
        System.IO.Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.ApplicationData), "AcerMode");

    public static readonly JsonSerializerOptions Json = new()
    {
        WriteIndented = true,

        try
        {
            var s = JsonSerializer.Deserialize<AppSettings>(File.ReadAllText(path), Json) ?? throw new JsonException("Arquivo vazio.");

            return (s, null);
```

```csharp
namespace AcerMode.Core;

public sealed class AppLog
{
    readonly long maxBytes;
    readonly object gate = new();
    public string Path { get; }

    public AppLog(string path, long maxBytes = 512 * 1024) { Path = path; this.maxBytes = maxBytes; }

    public void Write(string message)
    {
        lock (gate)
        {
            try
            {
                Directory.CreateDirectory(System.IO.Path.GetDirectoryName(Path)!);
                if (File.Exists(Path) && new FileInfo(Path).Length > maxBytes)
                    File.Move(Path, System.IO.Path.ChangeExtension(Path, null) + ".1.log", overwrite: true);
                File.AppendAllText(Path, $"{DateTime.Now:yyyy-MM-dd HH:mm:ss.fff} {message}\r\n");
            }
            catch (IOException) { }   // log nunca derruba o app
        }
    }

    public IReadOnlyList<string> Tail(int n)
    {
        lock (gate)
        {
            if (!File.Exists(Path)) return Array.Empty<string>();
            var lines = File.ReadAllLines(Path);
            return lines.Skip(Math.Max(0, lines.Length - n)).ToArray();
        }
    }
}
```

```csharp
// Troca de modo como transação: modo térmico → iluminação → ventoinhas.
// Se o modo falha, nada mais é aplicado; falha de iluminação/ventoinha vira aviso.
public sealed class ModeService

    readonly DirtyMarker fansDirty;
    readonly AppLog? log;

    public ModeService(IAcerHardware hw, Func<AppSettings> settings, DirtyMarker fansDirty, AppLog? log = null)

    public FanSettings ActiveFans { get; private set; } = new();
    public event Action<ThermalMode>? ModeChanged;

    public async Task<ThermalMode?> RefreshAsync()
    {
        int code = await hw.GetModeCodeAsync();

        return Active;

        await hw.SetModeAsync(mode);

    {

        if (current == ThermalMode.Turbo) return await ApplyModeAsync(beforeTurbo);
        beforeTurbo = current ?? ThermalMode.Balanced;
        return await ApplyModeAsync(ThermalMode.Turbo);
    }

        return w;

        await hw.SetFansAsync(auto);
        ActiveFans = auto;
        fansDirty.Clear();
        log?.Write("ventoinhas -> Auto");
    }

    public async Task RecoverAfterCrashAsync()
    {
        if (!fansDirty.Exists) return;
        log?.Write("ventoinhas em Personalizado na última sessão; voltando ao automático");
        await RevertFansToAutoAsync();
```

```csharp
namespace AcerMode.Core;

// Lê sensores na cadência certa e alimenta o FanGuard.
public sealed class SensorMonitor : IDisposable
{
    readonly IAcerHardware hw;
    readonly Func<FanSettings> activeFans;
    readonly CancellationTokenSource cts = new();

    public SensorMonitor(IAcerHardware hw, Func<FanSettings> activeFans) { this.hw = hw; this.activeFans = activeFans; }

    public bool UiVisible { get; set; }
    public SensorHistory History { get; } = new(120);
    public event Action<SensorReading>? Updated;
    public event Action<FanGuardAction, SensorReading?>? GuardTriggered;

    // 1 s com UI visível; 3 s escondido se há ventoinha em Personalizado; senão parado.
    public static bool ShouldPoll(long tick, bool visible, bool custom) => visible || (custom && tick % 3 == 0);

    public void Start() => _ = LoopAsync();

    async Task LoopAsync()
    {
        using var timer = new PeriodicTimer(TimeSpan.FromSeconds(1));
        long tick = 0;
        try { while (await timer.WaitForNextTickAsync(cts.Token)) await TickAsync(++tick); }
        catch (OperationCanceledException) { }
    }

    public async Task TickAsync(long tick)
    {
        var fans = activeFans();
        if (!ShouldPoll(tick, UiVisible, fans.Mode == FanMode.Custom)) return;
        SensorReading? r = null;
        try
        {
            r = await hw.ReadSensorsAsync();
            History.Add(r);
            Updated?.Invoke(r);
        }
        catch (HardwareException) { }
        var action = FanGuard.Evaluate(fans, r);
        if (action != FanGuardAction.None) GuardTriggered?.Invoke(action, r);
    }

    public void Dispose() => cts.Cancel();
}
```

```csharp
using System.Collections.Concurrent;
using System.Management;

namespace AcerMode.Core;

// Uma thread executa as chamadas à BIOS, uma por vez (chamadas simultâneas travam a BIOS,
// principalmente ao voltar da suspensão). Se uma chamada passa do limite, a thread é
// abandonada, outra assume e as próximas falham na hora durante a pausa.
public sealed class SerialWorker : IDisposable
{
    readonly Func<long> now;
    readonly long backoffMs;
    readonly object gate = new();
    BlockingCollection<Action> queue = null!;
    long failUntil = long.MinValue;

    public SerialWorker(Func<long>? clock = null, long backoffMs = 30_000)
    {
        now = clock ?? (() => Environment.TickCount64);
        this.backoffMs = backoffMs;
        StartThread();
    }

    void StartThread()
    {
        var q = new BlockingCollection<Action>();
        var t = new Thread(() => { foreach (var a in q.GetConsumingEnumerable()) a(); }) { IsBackground = true, Name = "AcerWmi" };
        t.Start();
        queue = q;
    }

    public async Task<T> Run<T>(Func<T> f, TimeSpan timeout)
    {
        BlockingCollection<Action> q;
        lock (gate)
        {
            if (now() < failUntil) throw new HardwareException("A BIOS não respondeu. Nova tentativa em instantes.");
            q = queue;
        }
        var tcs = new TaskCompletionSource<T>(TaskCreationOptions.RunContinuationsAsynchronously);
        q.Add(() =>
        {
            if (tcs.Task.IsCompleted) return;
            try { tcs.TrySetResult(f()); }
            catch (HardwareException ex) { tcs.TrySetException(ex); }
            catch (Exception ex) { tcs.TrySetException(new HardwareException(ex.Message)); }
        });
        if (await Task.WhenAny(tcs.Task, Task.Delay(timeout)) != tcs.Task)
        {
            lock (gate)
            {
                failUntil = now() + backoffMs;
                if (ReferenceEquals(q, queue)) { q.CompleteAdding(); StartThread(); }
            }
            tcs.TrySetException(new HardwareException("A BIOS não respondeu."));
        }
        return await tcs.Task;
    }

    public void Dispose() { lock (gate) queue.CompleteAdding(); }
}

public sealed class AcerWmi : IAcerHardware, IDisposable
{
    static readonly TimeSpan Timeout = TimeSpan.FromSeconds(4);
    readonly SerialWorker worker = new();
    ManagementObject? obj;   // só tocado pela thread do worker

    ManagementObject Obj()
    {
        if (obj is not null) return obj;
        using var s = new ManagementObjectSearcher(@"root\WMI", "SELECT * FROM AcerGamingFunction");
        foreach (ManagementObject o in s.Get()) return obj = o;
        throw new HardwareException("Interface AcerGamingFunction não encontrada.");
    }

    ulong Call(string method, object input)
    {
        try
        {
            var o = Obj();
            var p = o.GetMethodParameters(method);
            p["gmInput"] = input;
            var r = o.InvokeMethod(method, p, null);
            return Convert.ToUInt64(r["gmOutput"]);
        }
        catch (ManagementException) { obj = null; throw; }
    }

    void SetChecked(string method, object input, string what)
    {
        ulong status = Call(method, input);
        if (!Encoders.StatusOk(status)) throw new HardwareException($"A BIOS recusou {what} (status {status}).");
    }

    Task<T> Run<T>(Func<T> f) => worker.Run(f, Timeout);
    Task Run(Action a) => worker.Run(() => { a(); return true; }, Timeout);

    public Task<int> GetModeCodeAsync() => Run(() => Encoders.ThermalParse(Call("GetGamingMiscSetting", Encoders.ThermalSetting)));

    public Task SetModeAsync(ThermalMode mode) => Run(() => SetChecked("SetGamingMiscSetting", Encoders.ThermalSet(mode), "a troca de modo"));

    public Task ApplyLightingAsync(LightingProfile profile) => Run(() =>
    {

        if (p.Effect == KbEffect.Static)
            for (int z = 0; z < LightingProfile.ZoneCount; z++)
                SetChecked("SetGamingRgbKb", Encoders.ZoneColor(z, p.Zones[z]), "a cor da zona " + (z + 1));

        return new SensorReading(Read(Sensor.CpuTemp), Read(Sensor.GpuTemp), Read(Sensor.SystemTemp),
            Read(Sensor.CpuFanRpm), Read(Sensor.GpuFanRpm), DateTime.Now);

        if (back is not { } b || b.Cpu != f.Mode || b.Gpu != f.Mode)

            catch (Exception ex) { lines.Add($"{m}(0x{Convert.ToUInt64(input):X}) ERRO {ex.Message}"); }

        for (uint z = 1; z <= 8; z <<= 1) Q("GetGamingRgbKb", z);
        return lines;
    });

    public void Dispose() => worker.Dispose();

    }
```

```csharp
// Theme/AccentManager.cs
using System.Windows;
using System.Windows.Media;
using System.Windows.Media.Animation;
using AcerMode.Core;

namespace AcerMode.App.Theme;

// Uma cor de destaque para o app inteiro, que acompanha o modo ativo com cross-fade.
public static class AccentManager
{
    public static Color For(ThermalMode m) => m switch
    {
        ThermalMode.Quiet => Color.FromRgb(0x22, 0xD3, 0xEE),
        ThermalMode.Performance => Color.FromRgb(0x8B, 0x5C, 0xF6),
        ThermalMode.Turbo => Color.FromRgb(0xFF, 0x4D, 0x1F),
        ThermalMode.Eco => Color.FromRgb(0x22, 0xC5, 0x5E),
        _ => Color.FromRgb(0x3B, 0x82, 0xF6),
    };

    static readonly SolidColorBrush accent = new(For(ThermalMode.Balanced));
    static readonly SolidColorBrush soft = new(WithAlpha(For(ThermalMode.Balanced), 0x29));
    static readonly SolidColorBrush glow = new(WithAlpha(For(ThermalMode.Balanced), 0x8C));

    public static void Install(ResourceDictionary r)
    {
        r["AccentBrush"] = accent; r["AccentSoftBrush"] = soft; r["AccentGlowBrush"] = glow;
    }

    static Color WithAlpha(Color c, byte a) => Color.FromArgb(a, c.R, c.G, c.B);

    public static void Apply(ThermalMode m, bool animate)
    {
        var c = For(m);
        Fade(accent, c, animate); Fade(soft, WithAlpha(c, 0x29), animate); Fade(glow, WithAlpha(c, 0x8C), animate);
    }

    static void Fade(SolidColorBrush b, Color to, bool animate)
    {
        int ms = animate ? Motion.Ms(300) : 0;
        if (ms == 0) { b.BeginAnimation(SolidColorBrush.ColorProperty, null); b.Color = to; return; }
        b.BeginAnimation(SolidColorBrush.ColorProperty, new ColorAnimation(to, TimeSpan.FromMilliseconds(ms)) { EasingFunction = Motion.EaseOut });
    }
}
```

```csharp
using System.Windows;
using System.Windows.Media.Animation;

namespace AcerMode.App.Theme;

public static class Motion
{
    public static bool Enabled => SystemParameters.ClientAreaAnimation;
    public static int Ms(int ms) => Enabled ? ms : 0;
    public static readonly IEasingFunction EaseOut = new CubicEase { EasingMode = EasingMode.EaseOut };
    public static readonly IEasingFunction Spring = new BackEase { EasingMode = EasingMode.EaseOut, Amplitude = 0.3 };

    public static DoubleAnimation To(double to, int ms, IEasingFunction? ease = null) =>
        new(to, TimeSpan.FromMilliseconds(Ms(ms))) { EasingFunction = ease ?? EaseOut };
}
```

```csharp
// Theme/Backdrop.cs
using System.Runtime.InteropServices;
using System.Windows;
using System.Windows.Interop;
using System.Windows.Media;

namespace AcerMode.App.Theme;

public enum BackdropKind { Mica = 2, Acrylic = 3 }

// Mica/acrílico do Windows 11 numa janela WPF sem borda (WindowChrome com GlassFrameThickness -1).
public static class Backdrop
{
    [DllImport("dwmapi.dll")] static extern int DwmSetWindowAttribute(IntPtr hwnd, int attr, ref int value, int size);
    const int DarkMode = 20, CornerPreference = 33, SystemBackdrop = 38;

    public static void Apply(Window w, BackdropKind kind)
    {
        var h = new WindowInteropHelper(w).EnsureHandle();
        int on = 1, round = 2, k = (int)kind;
        DwmSetWindowAttribute(h, DarkMode, ref on, 4);
        DwmSetWindowAttribute(h, CornerPreference, ref round, 4);
        DwmSetWindowAttribute(h, SystemBackdrop, ref k, 4);
        if (HwndSource.FromHwnd(h) is { CompositionTarget: { } ct }) ct.BackgroundColor = Colors.Transparent;
    }
}
```

```csharp
// Startup/Elevation.cs
using System.Diagnostics;
using System.Security.Principal;

namespace AcerMode.App.Startup;

public static class Elevation
{
    public static bool IsAdmin => new WindowsPrincipal(WindowsIdentity.GetCurrent()).IsInRole(WindowsBuiltInRole.Administrator);

    // Relança o próprio exe pedindo UAC. Devolve false se a pessoa recusou.
    public static bool Relaunch(string[] args)
    {
        try
        {
            Process.Start(new ProcessStartInfo(Environment.ProcessPath!, string.Join(' ', args.Select(a => '"' + a + '"'))) { UseShellExecute = true, Verb = "runas" });
            return true;
        }
        catch (System.ComponentModel.Win32Exception) { return false; }
    }
}
```

```csharp
// Startup/SingleInstance.cs
namespace AcerMode.App.Startup;

// Abrir de novo só traz a janela da instância que já está na bandeja.
public sealed class SingleInstance : IDisposable
{
    readonly Mutex mutex;
    readonly EventWaitHandle showEvent;
    public bool IsFirst { get; }

    public SingleInstance(string name)
    {
        mutex = new Mutex(true, @"Local\" + name, out bool created);
        IsFirst = created;
        showEvent = new EventWaitHandle(false, EventResetMode.AutoReset, @"Local\" + name + "Show");
    }

    public void SignalFirst() => showEvent.Set();

    public void Listen(Action onShow) =>
        new Thread(() => { while (true) { showEvent.WaitOne(); onShow(); } }) { IsBackground = true }.Start();

    public void Dispose() { if (IsFirst) mutex.ReleaseMutex(); mutex.Dispose(); showEvent.Dispose(); }
}
```
